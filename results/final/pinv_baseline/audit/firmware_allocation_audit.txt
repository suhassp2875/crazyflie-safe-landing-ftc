static uint16_t capMinThrust(float thrust, uint32_t minThrust) {
  if (thrust < minThrust) {
    return minThrust;
  }

  return thrust;
}

static void powerDistributionLegacy(const control_t *control, motors_thrust_uncapped_t* motorThrustUncapped)
{
  int16_t r = control->roll / 2.0f;
  int16_t p = control->pitch / 2.0f;

  motorThrustUncapped->motors.m1 = control->thrust - r + p + control->yaw;
  motorThrustUncapped->motors.m2 = control->thrust - r - p - control->yaw;
  motorThrustUncapped->motors.m3 = control->thrust + r - p + control->yaw;
  motorThrustUncapped->motors.m4 = control->thrust + r + p - control->yaw;
}

static void powerDistributionForceTorque(const control_t *control, motors_thrust_uncapped_t* motorThrustUncapped) {
  static float motorForces[STABILIZER_NR_OF_MOTORS];

  const float arm = 0.707106781f * armLength;
  const float rollPart = 0.25f / arm * control->torqueX;
  const float pitchPart = 0.25f / arm * control->torqueY;
  const float thrustPart = 0.25f * control->thrustSi; // N (per rotor)
  const float yawPart = 0.25f * control->torqueZ / thrustToTorque;

  motorForces[0] = thrustPart - rollPart - pitchPart - yawPart;
  motorForces[1] = thrustPart - rollPart + pitchPart + yawPart;
  motorForces[2] = thrustPart + rollPart + pitchPart - yawPart;
  motorForces[3] = thrustPart + rollPart - pitchPart + yawPart;

  for (int motorIndex = 0; motorIndex < STABILIZER_NR_OF_MOTORS; motorIndex++) {
    float motorForce = motorForces[motorIndex];
    if (motorForce < 0.0f) {
      motorForce = 0.0f;
    }

    float motor_pwm = (-pwmToThrustB + sqrtf(pwmToThrustB * pwmToThrustB + 4.0f * pwmToThrustA * motorForce)) / (2.0f * pwmToThrustA);
    motorThrustUncapped->list[motorIndex] = motor_pwm * UINT16_MAX;
  }
}

void powerDistribution(const control_t *control, motors_thrust_uncapped_t* motorThrustUncapped)
{
  switch (control->controlMode) {
    case controlModeLegacy:
      powerDistributionLegacy(control, motorThrustUncapped);
      break;
    case controlModeForceTorque:
      powerDistributionForceTorque(control, motorThrustUncapped);
      break;
    case controlModeForce:
      powerDistributionForce(control, motorThrustUncapped);
      break;
    default:
      // Nothing here
      break;
  }
}

bool powerDistributionCap(const motors_thrust_uncapped_t* motorThrustBatCompUncapped, motors_thrust_pwm_t* motorPwm)
{
  const int32_t maxAllowedThrust = UINT16_MAX;
  bool isCapped = false;

  // Find highest thrust
  int32_t highestThrustFound = 0;
  for (int motorIndex = 0; motorIndex < STABILIZER_NR_OF_MOTORS; motorIndex++)
  {
    const int32_t thrust = motorThrustBatCompUncapped->list[motorIndex];
    if (thrust > highestThrustFound)
    {
      highestThrustFound = thrust;
    }
  }

  int32_t reduction = 0;
  if (highestThrustFound > maxAllowedThrust)
  {
    reduction = highestThrustFound - maxAllowedThrust;
    isCapped = true;
  }

  for (int motorIndex = 0; motorIndex < STABILIZER_NR_OF_MOTORS; motorIndex++)
  {
    int32_t thrustCappedUpper = motorThrustBatCompUncapped->list[motorIndex] - reduction;
    motorPwm->list[motorIndex] = capMinThrust(thrustCappedUpper, idleThrust);
  }

  return isCapped;
}

// SITL actuator fault injection parameters.
// enable = 0: healthy motors
// enable = 1: apply loss-of-effectiveness to selected motor
// motor  = 1, 2, 3, or 4
// eta    = motor effectiveness multiplier, clipped to [0, 1]
//          eta = 1.0 means healthy, eta = 0.6 means 40% loss.
static uint8_t sitlFaultEnable = 0;
static uint8_t sitlFaultMotor = 1;
static float sitlFaultEta = 1.0f;

static uint8_t sitlFtcEnable = 0;
static float sitlFtcHealthyBoost = 0.0f;
static float sitlFtcResidualM1 = 0.0f;
static float sitlFtcResidualM2 = 0.0f;
static float sitlFtcResidualM3 = 0.0f;
static float sitlFtcResidualM4 = 0.0f;

static uint16_t applySitlMotorFault(uint8_t motorIndex, uint16_t pwm)
{
  if (!sitlFaultEnable) {
    return pwm;
  }

  if (sitlFaultMotor != motorIndex) {
    return pwm;
  }

  float eta = sitlFaultEta;

  if (eta < 0.0f) {
    eta = 0.0f;
  }

  if (eta > 1.0f) {
    eta = 1.0f;
  }

  return (uint16_t)((float)pwm * eta);
}


static uint16_t clampSitlPwmFloat(float pwm)
{
  if (pwm < 0.0f) {
    return 0;
  }

  if (pwm > 65535.0f) {
    return 65535;
  }

  return (uint16_t)pwm;
}

static uint16_t applySitlFtcHealthyBoost(uint8_t motorIndex, uint16_t pwm)
{
  if (!sitlFtcEnable) {
    return pwm;
  }

  float residual = 0.0f;

  // Old behavior: equal boost to healthy motors only.
  if (motorIndex != sitlFaultMotor) {
    residual += sitlFtcHealthyBoost;
  }

  // New behavior: explicit per-motor residuals.
  // These are applied before the LoE fault model, so a residual on the
  // faulty motor is also degraded by sitlFaultEta.
  if (motorIndex == 1) {
    residual += sitlFtcResidualM1;
  } else if (motorIndex == 2) {
    residual += sitlFtcResidualM2;
  } else if (motorIndex == 3) {
    residual += sitlFtcResidualM3;
  } else if (motorIndex == 4) {
    residual += sitlFtcResidualM4;
  }

  return clampSitlPwmFloat((float)pwm + residual);
}


void motorsSetRatio(const motors_thrust_pwm_t* motorPwm)
{
  uint16_t m1 = motorPwm->motors.m1;
  uint16_t m2 = motorPwm->motors.m2;
  uint16_t m3 = motorPwm->motors.m3;
  uint16_t m4 = motorPwm->motors.m4;

  m1 = applySitlFtcHealthyBoost(1, m1);
  m2 = applySitlFtcHealthyBoost(2, m2);
  m3 = applySitlFtcHealthyBoost(3, m3);
  m4 = applySitlFtcHealthyBoost(4, m4);

  motorPower.m1 = applySitlMotorFault(1, m1);
  motorPower.m2 = applySitlMotorFault(2, m2);
  motorPower.m3 = applySitlMotorFault(3, m3);
  motorPower.m4 = applySitlMotorFault(4, m4);

  if (xTaskGetTickCount() - lastSentTime >= M2T(1)){
    memcpy(p.data , (uint8_t *) &motorPower , p.size);
    crtpSendPacket(&p);
    lastSentTime = xTaskGetTickCount();
  }
  // DEBUG_PRINT("%d , %d , %d , %d  \n", (int) motorPower.m1 , (int) motorPower.m2 , (int) motorPower.m3 , (int) motorPower.m4 );
}
